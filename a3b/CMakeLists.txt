if(GGML_CUDA)
  find_package(CUDAToolkit REQUIRED)
  enable_language(CUDA)
  add_library(strata_a3b_strata_native STATIC
    strata_native_experts.cu strata_prefill_mmq_adapter.cu
    ../src/prefill/moe_mmq.cu ../src/kernels/cuda/iq_kernels.cu
    ../src/kernels/cuda/fused_gdn.cu)
  # GGML's CUDA GDN and RMSNorm compile with -use_fast_math. Match their
  # transcendental and reciprocal semantics for the opt-in fused A3B step only.
  set_source_files_properties(../src/kernels/cuda/fused_gdn.cu
    PROPERTIES COMPILE_OPTIONS "-use_fast_math")
  set_source_files_properties(strata_prefill_mmq_adapter.cu ../src/prefill/moe_mmq.cu
    PROPERTIES COMPILE_OPTIONS "-use_fast_math;--extended-lambda")
  set_target_properties(strata_a3b_strata_native PROPERTIES
    CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON CUDA_RUNTIME_LIBRARY Shared)
  target_include_directories(strata_a3b_strata_native PRIVATE
    ../include ../src/kernels/cuda ../third_party/ggml
    "${LLAMA_SOURCE}/ggml/src" "${LLAMA_SOURCE}/ggml/src/ggml-cuda")
  # mmq::Context embeds ggml_backend_cuda_context from common.cuh. Its
  # conditional graph field must match the already-built ggml-cuda library.
  if(GGML_CUDA_GRAPHS)
    target_compile_definitions(strata_a3b_strata_native PRIVATE GGML_CUDA_USE_GRAPHS)
  endif()
  target_include_directories(strata_a3b_strata_native PUBLIC .)
  target_link_libraries(strata_a3b_strata_native PUBLIC ggml-base CUDA::cudart)
  target_sources(ggml-cuda PRIVATE
    "${LLAMA_SOURCE}/ggml/src/ggml-cuda/strata-native-iq.cu")
  target_include_directories(ggml-cuda PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../include")
  target_link_libraries(ggml-cuda PRIVATE strata_a3b_strata_native)
  target_compile_definitions(strata_a3b_backend PRIVATE STRATA_A3B_STRATA_NATIVE=1)
  target_link_libraries(strata_a3b_backend PRIVATE strata_a3b_strata_native ggml-cuda)
  add_executable(strata-moe-iq3s-fused-gu-parity iq3s_fused_gu_parity.cpp)
  target_include_directories(strata-moe-iq3s-fused-gu-parity PRIVATE ../include ../third_party/ggml)
  target_link_libraries(strata-moe-iq3s-fused-gu-parity PRIVATE strata_a3b_strata_native CUDA::cudart)
  add_executable(strata-moe-expert-plan-parity expert_plan_parallel_test.cu)
  set_target_properties(strata-moe-expert-plan-parity PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(strata-moe-expert-plan-parity PRIVATE CUDA::cudart)
  add_executable(strata-moe-prefill-mmq-probe strata_prefill_mmq_probe.cpp)
  target_include_directories(strata-moe-prefill-mmq-probe PRIVATE ../include)
  target_link_libraries(strata-moe-prefill-mmq-probe PRIVATE ggml ggml-cuda CUDA::cudart)
  add_executable(strata-moe-prefill-mmq-adapter-parity strata_prefill_mmq_adapter_parity.cpp)
  target_include_directories(strata-moe-prefill-mmq-adapter-parity PRIVATE ../include)
  target_link_libraries(strata-moe-prefill-mmq-adapter-parity PRIVATE
    ggml ggml-cuda strata_a3b_strata_native CUDA::cudart)
endif()

if(GGML_CUDA)
  # Opt-in profiling uses NVTX v3's header-only API, with a source-level
  # __has_include guard for CUDA toolkits that do not ship its headers.
  target_compile_definitions(strata_a3b_speculative PRIVATE STRATA_A3B_NVTX_CUDA=1)
  target_compile_definitions(strata-moe PRIVATE STRATA_A3B_NVTX_CUDA=1)
  target_include_directories(strata_a3b_speculative PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
  target_include_directories(strata-moe PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
  if(UNIX)
    # NVTX's header-only Linux implementation loads the profiler through dlopen.
    target_link_libraries(strata_a3b_speculative PRIVATE ${CMAKE_DL_LIBS})
    target_link_libraries(strata-moe PRIVATE ${CMAKE_DL_LIBS})
  endif()
endif()
